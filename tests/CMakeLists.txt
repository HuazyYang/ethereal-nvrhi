# ABI lint (abi_lint.py): the public interfaces follow the rules that make MSVC and MinGW agree on them (no class by
# value in virtual signatures, no virtual destructors, no overloaded virtuals, noexcept, no bases on data structs,
# C exports only). The self test plants one violation of each rule in a copy of the headers.
find_package(Python3 COMPONENTS Interpreter QUIET)

    add_test(NAME nvrhi_abi_lint
        COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/abi_lint.py ${PROJECT_SOURCE_DIR}/include/nvrhi)
    add_test(NAME nvrhi_abi_lint_selftest
        COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/abi_lint.py --self-test ${PROJECT_SOURCE_DIR}/include/nvrhi)
else()
    message(STATUS "NVRHI: Python 3 not found, skipping the ABI lint (nvrhi_abi_lint)")
endif()

# One compile-only translation unit per public header, which includes that header alone (twice, for the include
# guard): every public header is self-contained. Building nvrhi_header_tus is the test.
file(GLOB_RECURSE nvrhi_public_headers RELATIVE "${PROJECT_SOURCE_DIR}/include"
    "${PROJECT_SOURCE_DIR}/include/nvrhi/*.h")
list(SORT nvrhi_public_headers)
set(nvrhi_header_tus "")
foreach(header IN LISTS nvrhi_public_headers)
    if((header STREQUAL "nvrhi/d3d11.h" AND NOT NVRHI_WITH_DX11) OR
       (header STREQUAL "nvrhi/d3d12.h" AND NOT NVRHI_WITH_DX12) OR
       (header STREQUAL "nvrhi/vulkan.h" AND NOT NVRHI_WITH_VULKAN))
        continue()
    endif()
    string(REGEX REPLACE "[/.]" "_" tu_name "${header}")
    set(tu "${CMAKE_CURRENT_BINARY_DIR}/header_tus/${tu_name}.cpp")
    set(tu_content "// Generated by tests/CMakeLists.txt: ${header} compiles on its own.\n#include <${header}>\n#include <${header}>\n")
    if(EXISTS "${tu}")
        file(READ "${tu}" tu_old)
    else()
        set(tu_old "")
    endif()
    if(NOT tu_old STREQUAL tu_content)
        file(WRITE "${tu}" "${tu_content}")
    endif()
    list(APPEND nvrhi_header_tus "${tu}")
endforeach()
add_library(nvrhi_header_tus OBJECT ${nvrhi_header_tus})
target_link_libraries(nvrhi_header_tus PRIVATE nvrhi)  # its usage requirements only: nothing is linked
if(NVRHI_WITH_DX12)
    target_link_libraries(nvrhi_header_tus PRIVATE ${nvrhi_directx_headers_target})
endif()
if(NVRHI_WITH_VULKAN)
    target_link_libraries(nvrhi_header_tus PRIVATE ${nvrhi_vulkan_headers_target})
endif()
set_target_properties(nvrhi_header_tus PROPERTIES FOLDER "NVRHI/Tests")

# The cross-module foundation.h test (core/cross_module_exe.cpp): a test DLL on the DLL CRT and a test EXE on the
# static CRT create COM objects with opposite NVRHI_PACK_CONTROL_BLOCK_AND_OBJECT values and release each
# other's objects, and share containers and blobs through nvrhi_core's heap. Built once per packing mode of the
# DLL (the EXE uses the other one).
if(WIN32)
    foreach(dll_pack 0 1)
        if(dll_pack)
            set(exe_pack 0)
        else()
            set(exe_pack 1)
        endif()
        set(xm_dll nvrhi_test_cross_module_dll_p${dll_pack})
        set(xm_exe nvrhi_test_cross_module_p${dll_pack})

        add_library(${xm_dll} SHARED core/cross_module_dll.cpp core/cross_module.h)
        target_link_libraries(${xm_dll} PRIVATE nvrhi_core)
        target_compile_definitions(${xm_dll} PRIVATE NVRHI_PACK_CONTROL_BLOCK_AND_OBJECT=${dll_pack})
        set_target_properties(${xm_dll} PROPERTIES
            MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL"
            FOLDER "NVRHI/Tests")

        add_executable(${xm_exe} core/cross_module_exe.cpp core/cross_module.h)
        target_link_libraries(${xm_exe} PRIVATE ${xm_dll} nvrhi_core)
        target_compile_definitions(${xm_exe} PRIVATE NVRHI_PACK_CONTROL_BLOCK_AND_OBJECT=${exe_pack})
        set_target_properties(${xm_exe} PROPERTIES
            MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>"
            FOLDER "NVRHI/Tests")
        add_test(NAME ${xm_exe} COMMAND ${xm_exe})
        if(CMAKE_VERSION VERSION_GREATER_EQUAL 3.22)
            # The test DLL, wherever the build puts it (nvrhi_core: below).
            set_tests_properties(${xm_exe} PROPERTIES ENVIRONMENT_MODIFICATION
                "PATH=path_list_prepend:$<TARGET_FILE_DIR:${xm_dll}>")
        endif()
    endforeach()
endif()
