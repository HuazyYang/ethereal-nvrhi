cmake_minimum_required(VERSION 3.22)
project(NvrhiInstalledConsumer LANGUAGES CXX)
option(TEST_ARTIFACT_CONTRACT "Also consume the independent artifact reader" ON)
find_package(nvrhi CONFIG REQUIRED)
add_executable(nvrhi_package_smoke main.cpp)
target_link_libraries(nvrhi_package_smoke PRIVATE nvrhi nvrhi::cuda)
enable_testing()
add_test(NAME installed_package_api COMMAND nvrhi_package_smoke)
if(TEST_ARTIFACT_CONTRACT)
    find_package(ShaderTool CONFIG REQUIRED COMPONENTS Artifacts)
    add_executable(nvrhi_package_contract ../shader-program.cpp)
    target_link_libraries(nvrhi_package_contract PRIVATE nvrhi ShaderTool::Artifacts)
    add_test(NAME installed_package_contract COMMAND nvrhi_package_contract)
endif()
foreach(compiler_target ShaderTool::ShaderTool ShaderTool::slang ShaderTool::dxcompiler)
    if(TARGET ${compiler_target})
        message(FATAL_ERROR "Runtime package imported compiler dependency ${compiler_target}")
    endif()
endforeach()
if(WIN32)
    get_property(package_tests DIRECTORY PROPERTY TESTS)
    set_tests_properties(${package_tests} PROPERTIES ENVIRONMENT_MODIFICATION
        "PATH=path_list_prepend:$<TARGET_FILE_DIR:nvrhi_core>;PATH=path_list_prepend:$<TARGET_FILE_DIR:nvrhi>")
endif()
